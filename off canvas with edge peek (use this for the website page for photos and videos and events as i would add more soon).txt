<script>
((root) => {
    const $ = (s) => root.querySelector(s);
    const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
    let hover = false, until = 0;
    const bump = () => (until = performance.now() + 2500);
    root.addEventListener('pointerenter', () => (hover = true));
    root.addEventListener('pointerleave', () => { hover = false; bump(); });
    root.addEventListener('pointerdown', bump);
    const free = () => !root.dataset.paused && !hover && performance.now() > until;
    const cur = $('.cur');
    const wait = async (ms) => {
      await sleep(ms);
      while (root.isConnected && !free()) { cur.classList.add('off'); await sleep(250); }
      if (!root.isConnected) throw 0;
    };
    const at = (el, fx = 0.5, fy = 0.5) => {
      const r = el.getBoundingClientRect(), b = root.getBoundingClientRect(), k = b.width / root.offsetWidth || 1;
      cur.style.transform = `translate(${(r.left - b.left + r.width * fx) / k - 4}px,${(r.top - b.top + r.height * fy) / k - 4}px)`;
      cur.classList.remove('off');
    };
    const tap = async () => { cur.classList.add('press'); await sleep(120); cur.classList.remove('press'); };
    const mn = $('.mn'), ttl = $('.ttl'), tl = $('.tl');
    const nav = (it) => {
      if (!mn || it.classList.contains('on')) return;
      root.querySelectorAll('.it.on').forEach((x) => x.classList.remove('on'));
      it.classList.add('on');
      mn.classList.add('out');
      setTimeout(() => { ttl.textContent = it.dataset.t; tl.dataset.v = it.dataset.v; mn.classList.remove('out'); }, 150);
    };
    root.addEventListener('click', (e) => { const it = e.target.closest && e.target.closest('.it[data-t]'); if (it) nav(it); });
    const app = $('.app'), pn = $('.pn'), ez = $('.ez'), cl = $('.cl'), fb = $('.fb');
    const tiles = [...root.querySelectorAll('.tl i')];
    let t = 0;
    const off = (v) => { clearTimeout(t); app.classList.remove('peek'); app.classList.toggle('off', v); };
    const peek = (v) => {
      clearTimeout(t);
      if (!app.classList.contains('off')) return;
      if (v) app.classList.add('peek');
      else t = setTimeout(() => app.classList.remove('peek'), 300);
    };
    cl.addEventListener('click', () => off(!app.classList.contains('peek')));
    fb.addEventListener('click', () => off(false));
    [ez, pn].forEach((el) => {
      el.addEventListener('pointerenter', () => peek(true));
      el.addEventListener('pointerleave', () => peek(false));
    });
    const main = async () => {
      for (;;) {
        await wait(1200);
        at(cl);
        await wait(800);
        await tap();
        off(true);
        await wait(1500);
        at(ez, 0.5, 0.45);
        await wait(900);
        peek(true);
        await wait(1700);
        at(tiles[4]);
        peek(false);
        await wait(1500);
        at(fb);
        await wait(800);
        await tap();
        off(false);
        await wait(1500);
        cur.classList.add('off');
        await wait(900);
      }
    };
    main().catch(() => {});
  })(document.querySelector('.d'));
