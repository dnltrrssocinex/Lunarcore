        [roleEls, 1],
        [goalEls, 1],
        [goalEls, 3],
        [goalEls, 4],
        [[cont], 0],
      ];
      (async () => {
        await wait(1800);
        while (root.isConnected) {
          if (!idle() || busy) {
            await wait(400);
            continue;
          }
          role = 2;
          picked.clear();
          picked.add(0);
          render();
          for (const [list, i] of plan) {
            if (!(await tap(list[i], 0.5, 0.5))) break;
            await wait(list === goalEls || list === roleEls ? 650 : 3600);
          }
          await wait(800);
        }
      })();
    })(document.querySelector('.d'));
